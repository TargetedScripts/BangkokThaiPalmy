${s.type === 'food' ? `
${s.promoTag}

${s.id}: ${s.title}

$${s.price.toFixed(2)}

` : ''}
`).join(''); if (slidesData.length > 1) { setInterval(() => { const slides = container.querySelectorAll('.hero-slide'); slides[heroIndex].classList.replace('opacity-100', 'opacity-0'); heroIndex = (heroIndex + 1) % slides.length; slides[heroIndex].classList.replace('opacity-0', 'opacity-100'); }, 4000); } } // Render Menu Grid dengan Custom Promo Tag Badge function renderMenu() { if (!activeConfig.menuItems) return; const items = currentCategory === 'ALL' ? activeConfig.menuItems : activeConfig.menuItems.filter(i => i.category === currentCategory); document.getElementById('menuContainer').innerHTML = items.map(item => { const isSoldOut = item.isAvailable === false || item.stock <= 0; const hasPromo = item.promoPrice && item.promoPrice < item.price; const finalPrice = hasPromo ? item.promoPrice : item.price; const promoTag = item.promoTag || (hasPromo ? "SPECIAL OFFER" : ""); return `
${item.image ? `` : `
${activeConfig.store.name}
`} ${isSoldOut ? `
SOLD OUT
` : ''} ${promoTag && !isSoldOut ? `
${promoTag}
` : ''}

${item.id}: ${item.name}

${hasPromo ? `
$${item.price.toFixed(2)}
$${finalPrice.toFixed(2)}
` : `
$${item.price.toFixed(2)}
`}

${item.description}

${getSpiceBadge(item.spiceLevel)}
${!isSoldOut ? `` : `` }
`; }).join(''); }
